country_event = { ## Communist Crystal Ponies # Elections
	id = crycom.3
	immediate = {log = "[GetDateText]: [Root.GetName]: event crycom.3"}
	title = crycom.3.t
	desc = { text = crycom.3.d_yes_com
		trigger = {
			communism > 0.499
		}
	}
	desc = { text = crycom.3.d_no_com
		trigger = {
			communism < 0.5
		}
	}
	picture = GFX_event_crystal_city
	is_triggered_only = yes
	trigger = {
		always = yes
	}
	option = {
		trigger = {
			communism > 0.499
		}
		name = crycom.3.a
		set_politics = { ruling_party = communism elections_allowed = no }
		set_cosmetic_tag = CRY_SOMIMP
		leave_faction = no
		hidden_effect = {
			random_list = { 50 = { add_political_power = 50 } 50 = { add_political_power = 100 } }
		}
	}
	option = {
		trigger = {
			communism < 0.5
		}
		name = crycom.3.b
		set_temp_variable = {
			civ_war_size = {
				value = party_popularity@communism
				add = 0.1
				clamp = {
					min = 0.4
					max = 0.6
				}
			}
		}
		start_civil_war = {
			ruling_party = communism
			ideology = democratic
			size = civ_war_size
		}
		set_cosmetic_tag = CRY_SOMIMP
		add_threat = 10
		leave_faction = no
		add_popularity = { ideology = communism popularity = 0.2 }
		random_other_country = {
			limit = { original_tag = CRY
			has_government = democratic }
			set_cosmetic_tag = CRY_DEM
			add_popularity = { ideology = democratic popularity = 0.3 }
			save_event_target_as = dem_CRY
		}
	}
}
