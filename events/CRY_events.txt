country_event = {
	id = CRY_depression.3
	title = CRY_depression.3.t
	desc = CRY_depression.3.d
	picture = GFX_event_sombra_shadow
	is_triggered_only = yes

	trigger = {
		tag = CRY
		NOT = { has_country_flag = CRY_sombra_event_3_fired }
	}

	immediate = {
		set_country_flag = CRY_sombra_event_3_fired
	}
	option = {
		name = CRY_depression.3.a # "Prepare for the worst"
		add_to_variable = { great_depression_var = 4 }
		add_stability = -0.07
		# Apply war support based on depression level
		if = {
			limit = { check_variable = { great_depression_var < 6 } }
			add_war_support = 0.10 # Low depression = ponies rally
		}
		else_if = {
			limit = { check_variable = { great_depression_var < 9 } }
			add_war_support = 0.05
		}
		else = {
			add_war_support = -0.05 # High depression = despair
		}
		CRY_update_ui_effect = yes
	}
}

# OUTCOME EVENTS (5 events)
# Determined by CRY_new_deal_help + great_depression_var
# Buff = civ speed + stability + Five Year Plan bonuses
# Debuff = mil speed - stability/war support + Five Year Plan penalties
###############################################

# Event 4 - First economic assessment (fires after event 1, ~210 days)
country_event = {
	id = CRY_depression.4
	title = CRY_depression.4.t
	desc = CRY_depression.4.d
	picture = GFX_event_crystal_heart
	is_triggered_only = yes

	trigger = { tag = CRY }

	immediate = {
		# Calculate outcome score: new_deal_help (primary) + depression modifier (secondary)
		set_temp_variable = { CRY_outcome_score = CRY_new_deal_help }
		# Depression penalty: high depression reduces score
		set_temp_variable = { CRY_depression_penalty = great_depression_var }
		divide_temp_variable = { CRY_depression_penalty = 3 }
		subtract_from_temp_variable = { CRY_outcome_score = CRY_depression_penalty }
	}

	# Good outcome (score >= 8)
	option = {
		name = CRY_depression.4.a_good
		trigger = { check_variable = { CRY_outcome_score >= 8 } }
		custom_effect_tooltip = CRY_good_outcome_tt
		add_stability = 0.05
		# Five Year Plan - civ construction bonus
		add_to_variable = { CRY_production_speed_industrial_complex_factor_variable = 0.05 }
		add_to_variable = { CRY_five_year_plan_base_variable = 0.02 }
		add_to_variable = { CRY_five_year_plan_gain_variable = 0.01 }
		CRY_decrease_crisis_level_by_5 = yes
		CRY_update_ui_effect = yes
		country_event = { id = CRY_depression.5 days = 120 }
	}

	# Mediocre outcome (score 4-7)
	option = {
		name = CRY_depression.4.a_ok
		trigger = {
			check_variable = { CRY_outcome_score >= 4 }
			check_variable = { CRY_outcome_score < 8 }
		}
		custom_effect_tooltip = CRY_ok_outcome_tt
		add_stability = 0.02
		add_to_variable = { CRY_production_speed_industrial_complex_factor_variable = 0.02 }
		CRY_update_ui_effect = yes
		country_event = { id = CRY_depression.5 days = 120 }
	}

	# Bad outcome (score < 4)
	option = {
		name = CRY_depression.4.a_bad
		trigger = { check_variable = { CRY_outcome_score < 4 } }
		custom_effect_tooltip = CRY_bad_outcome_tt
		add_stability = -0.05
		add_war_support = -0.05
		# Five Year Plan - mil focus instead
		add_to_variable = { CRY_production_speed_arms_factory_factor_variable = 0.05 }
		add_to_variable = { CRY_five_year_plan_consumer_goods_expected_value_variable = 0.05 }
		CRY_update_ui_effect = yes
		country_event = { id = CRY_depression.5 days = 120 }
	}
}
