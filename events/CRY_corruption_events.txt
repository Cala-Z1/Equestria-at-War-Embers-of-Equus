add_namespace = CRY_corruption

# ── Sombra's Influence Grows (corruption increases) ───────────────────────
country_event = {
	id = CRY_corruption.1
	title = CRY_corruption.1.t
	desc = CRY_corruption.1.d
	picture = GFX_report_event_crystal_empire

	is_triggered_only = yes
	hidden = no

	trigger = {
		tag = CRY
		NOT = { has_country_flag = CRY_sombra_defeated }
	}

	option = {
		name = CRY_corruption.1.a
		add_to_variable = { CRY_corruption_level = 10 }
		clamp_variable = {
			var = CRY_corruption_level
			min = 0
			max = 100
		}
		add_stability = -0.03
	}
}

# ── Shadow Spreads Through the Mines ─────────────────────────────────────
country_event = {
	id = CRY_corruption.2
	title = CRY_corruption.2.t
	desc = CRY_corruption.2.d
	picture = GFX_report_event_crystal_empire

	is_triggered_only = yes

	trigger = {
		tag = CRY
		NOT = { has_country_flag = CRY_sombra_defeated }
		check_variable = { CRY_corruption_level > 40 }
	}

	option = {
		name = CRY_corruption.2.a # Do nothing
		add_to_variable = { CRY_corruption_level = 8 }
		clamp_variable = { var = CRY_corruption_level min = 0 max = 100 }
		add_political_power = -30
	}

	option = {
		name = CRY_corruption.2.b # Dispatch guards
		trigger = { has_war = no }
		add_to_variable = { CRY_corruption_level = 3 }
		clamp_variable = { var = CRY_corruption_level min = 0 max = 100 }
		add_manpower = -500
		add_political_power = -50
	}
}

# ── Random monthly corruption increase ───────────────────────────────────
country_event = {
	id = CRY_corruption.4
	title = CRY_corruption.4.t
	desc = CRY_corruption.4.d
	picture = GFX_report_event_crystal_empire

	trigger = {
		tag = CRY
		NOT = { has_country_flag = CRY_sombra_defeated }
		check_variable = { CRY_corruption_level > 5 }
	}

	mean_time_to_happen = {
		days = 120
		modifier = {
			factor = 0.5
			check_variable = { CRY_corruption_level > 70 }
		}
		modifier = {
			factor = 2.0
			check_variable = { CRY_corruption_level < 30 }
		}
		modifier = {
			factor = 0.7
			has_war = yes
		}
	}

	option = {
		name = CRY_corruption.4.a
		add_to_variable = { CRY_corruption_level = 5 }
		clamp_variable = { var = CRY_corruption_level min = 0 max = 100 }
	}
}
