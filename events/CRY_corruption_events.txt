# Crystal Corruption Events

add_namespace = CRY_corruption

# Event 1 - Sombra Defeated / Crystal Heart Restored
country_event = {
	id = CRY_corruption.1
	title = CRY_corruption.1.t
	desc = CRY_corruption.1.d
	picture = GFX_event_crystal_heart

	is_triggered_only = yes

	trigger = {
		tag = CRY
		check_variable = { CRY_corruption_level < 1 }
		NOT = { has_country_flag = CRY_sombra_defeated }
	}

	option = {
		name = CRY_corruption.1.a
		CRY_sombra_defeated_effect = yes
		custom_effect_tooltip = CRY_sombra_defeated_tt
	}
}

# Event 2 - Critical Warning
country_event = {
	id = CRY_corruption.2
	title = CRY_corruption.2.t
	desc = CRY_corruption.2.d
	picture = GFX_event_sombra_shadow

	is_triggered_only = yes

	trigger = {
		tag = CRY
		check_variable = { CRY_corruption_level > 74 }
		NOT = { has_country_flag = CRY_sombra_defeated }
	}

	option = {
		name = CRY_corruption.2.a
		add_stability = -0.05
		custom_effect_tooltip = CRY_critical_warning_tt
	}
}

# Event 3 - Max Corruption debuffs kick in
country_event = {
	id = CRY_corruption.3
	title = CRY_corruption.3.t
	desc = CRY_corruption.3.d
	picture = GFX_event_sombra_shadow

	is_triggered_only = yes

	trigger = {
		tag = CRY
		check_variable = { CRY_corruption_level > 99 }
		NOT = { has_country_flag = CRY_sombra_defeated }
	}

	option = {
		name = CRY_corruption.3.a
		add_stability = -0.10
		add_war_support = -0.10
		add_political_power = -50
		custom_effect_tooltip = CRY_max_corruption_tt
	}
}
