on_actions = {

	# Initialise corruption variable on game start
	on_startup = {
		effect = {
			if = {
				limit = { tag = CRY }
				set_variable = { CRY_corruption_level = 60 }
			}
		}
	}

	# Monthly tick — apply debuffs based on corruption level
	on_monthly_CRY = {
		effect = {
			if = {
				limit = {
					tag = CRY
					check_variable = { CRY_corruption_level > 79 }
					NOT = { has_country_flag = CRY_corruption_critical_active }
				}
				set_country_flag = CRY_corruption_critical_active
				clr_country_flag = CRY_corruption_high_active
				clr_country_flag = CRY_corruption_low_active
				add_ideas = CRY_sombra_influence_critical
				country_event = { id = CRY_corruption.3 days = 3 }
			}
			else_if = {
				limit = {
					tag = CRY
					check_variable = { CRY_corruption_level > 49 }
					NOT = { check_variable = { CRY_corruption_level > 79 } }
					NOT = { has_country_flag = CRY_corruption_high_active }
				}
				set_country_flag = CRY_corruption_high_active
				clr_country_flag = CRY_corruption_critical_active
				clr_country_flag = CRY_corruption_low_active
				remove_ideas = CRY_sombra_influence_critical
				add_ideas = CRY_sombra_influence_high
			}
			else_if = {
				limit = {
					tag = CRY
					check_variable = { CRY_corruption_level < 50 }
					NOT = { check_variable = { CRY_corruption_level < 1 } }
					NOT = { has_country_flag = CRY_corruption_low_active }
				}
				set_country_flag = CRY_corruption_low_active
				clr_country_flag = CRY_corruption_high_active
				clr_country_flag = CRY_corruption_critical_active
				remove_ideas = CRY_sombra_influence_high
				remove_ideas = CRY_sombra_influence_critical
				add_ideas = CRY_sombra_influence_low
			}
			else_if = {
				limit = {
					tag = CRY
					check_variable = { CRY_corruption_level < 1 }
					NOT = { has_country_flag = CRY_sombra_defeated }
				}
				set_country_flag = CRY_sombra_defeated
				remove_ideas = CRY_sombra_influence_low
				remove_ideas = CRY_sombra_influence_high
				remove_ideas = CRY_sombra_influence_critical
				add_ideas = CRY_crystal_heart_restored
				country_event = { id = CRY_corruption.10 days = 1 }
			}
		}
	}
}
