CRY_decrease_crisis_level_by_10 = {
	if = {
		limit = {
			check_variable = { CRY_crisis_level > 10 }
		}
		subtract_from_variable = { CRY_crisis_level = 10 }
	}
	else_if = {
		limit = {
			check_variable = { CRY_crisis_level < 10 }
		}
		set_variable = { CRY_crisis_level = 0 }
	}
}
